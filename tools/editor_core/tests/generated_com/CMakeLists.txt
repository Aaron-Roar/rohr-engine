cmake_minimum_required(VERSION 3.20)
project(rohr_generated_com LANGUAGES C)
find_package(Rohr CONFIG REQUIRED)
set(COM_GENERATED_DIR "" CACHE PATH "Directory generated by center_of_mass_persistence_test")
if(NOT EXISTS "${COM_GENERATED_DIR}/project_objects.c")
    message(FATAL_ERROR "Run center_of_mass_persistence_test and set COM_GENERATED_DIR to its src/generated directory")
endif()
add_executable(generated_com main.c "${COM_GENERATED_DIR}/project_objects.c")
target_include_directories(generated_com PRIVATE "${COM_GENERATED_DIR}")
target_link_libraries(generated_com PRIVATE Rohr::Engine)

enable_testing()
add_test(NAME generated_com COMMAND generated_com)
set_tests_properties(generated_com PROPERTIES
    ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
