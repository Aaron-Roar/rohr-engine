cmake_minimum_required(VERSION 3.20)
project(rohr_generated_areas LANGUAGES C)
find_package(Rohr CONFIG REQUIRED)
set(AREA_GENERATED_DIR "" CACHE PATH "Directory generated by editor_soft_areas_test")
if(NOT EXISTS "${AREA_GENERATED_DIR}/project_objects.c")
    message(FATAL_ERROR "Run editor_soft_areas_test and set AREA_GENERATED_DIR to its src/generated directory")
endif()
add_executable(generated_areas main.c "${AREA_GENERATED_DIR}/project_objects.c"
    "${AREA_GENERATED_DIR}/project_viewports.c")
target_include_directories(generated_areas PRIVATE "${AREA_GENERATED_DIR}")
target_link_libraries(generated_areas PRIVATE Rohr::Engine)
enable_testing()
add_test(NAME generated_areas COMMAND generated_areas)
set_tests_properties(generated_areas PROPERTIES
    ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
