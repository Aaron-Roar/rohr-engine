cmake_minimum_required(VERSION 3.20)
project(physics_collision_experiment LANGUAGES C)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${REPO}/examples/cmake/rohr_example.cmake")
rohr_example_bootstrap()
set(ROHR_EXPERIMENT_ENGINE_ARCHIVE "" CACHE FILEPATH "Optional frozen archive matching the installed SDK")
if(ROHR_EXPERIMENT_ENGINE_ARCHIVE)
    get_target_property(configs Rohr::Engine IMPORTED_CONFIGURATIONS)
    foreach(config IN LISTS configs)
        set_property(TARGET Rohr::Engine PROPERTY IMPORTED_LOCATION_${config} "${ROHR_EXPERIMENT_ENGINE_ARCHIVE}")
    endforeach()
endif()
enable_testing()
function(experiment_target target)
    target_sources(${target} PRIVATE experiment.c geometry.c audit.c)
    target_include_directories(${target} PRIVATE "${REPO}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    target_link_libraries(${target} PRIVATE ${ROHR_ENGINE_TARGET})
    target_link_options(${target} PRIVATE "LINKER:--wrap=physics_rigid_integrate"
        "LINKER:--wrap=physics_pipeline_joint_constraints_solve"
        "LINKER:--wrap=physics_pipeline_constraints_solve")
endfunction()
add_executable(collision_matrix matrix.c)
experiment_target(collision_matrix)
add_executable(geometry_checks geometry_checks.c geometry.c)
target_compile_options(geometry_checks PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
target_link_libraries(geometry_checks PRIVATE ${ROHR_ENGINE_TARGET})
add_test(NAME geometry_checks COMMAND geometry_checks)
add_executable(lifetime_checks lifetime_checks.c)
experiment_target(lifetime_checks)
add_test(NAME lifetime_checks COMMAND lifetime_checks)
add_executable(car_comparison "${REPO}/examples/soft-body/src/main.c"
    "${REPO}/examples/soft-body/src/car_controls.c"
    "${REPO}/examples/soft-body/src/generated/project_objects.c")
target_include_directories(car_comparison PRIVATE "${REPO}/examples/soft-body/src/generated")
rohr_add_example_runtime(car_comparison)
experiment_target(car_comparison)
rohr_stage_example_assets(car_comparison soft-body)
add_executable(pit_comparison "${REPO}/tests/particle_pit.c")
experiment_target(pit_comparison)
rohr_stage_example_assets(pit_comparison soft-body)
add_executable(flies_comparison flies.c)
experiment_target(flies_comparison)
rohr_stage_example_assets(flies_comparison flies-in-pit)
foreach(test physics_recovery collision_response particle_collision_mode force_preparation)
    add_executable(${test}_check "${REPO}/tests/${test}.c")
    experiment_target(${test}_check)
    add_test(NAME ${test} COMMAND ${test}_check)
endforeach()
